# =========================
# ЗАДАНИЕ 1. ФАЙЛОВАЯ СИСТЕМА
# =========================

# Создание каталогов
New-Item PShellLab4 -ItemType Directory
New-Item PShellLab4\DOCS -ItemType Directory
New-Item PShellLab4\VIDEO -ItemType Directory
New-Item PShellLab4\GRAPH -ItemType Directory
New-Item PShellLab4\TEXT -ItemType Directory
New-Item PShellLab4\DOCX -ItemType Directory
New-Item PShellLab4\TXT -ItemType Directory

# Просмотр содержимого
Get-ChildItem

# Создание файлов
1..8 | % {New-Item "PShellLab4\TEXT\$_.txt" -ItemType File}

# Копирование файлов
Copy-Item PShellLab4\TEXT\7.txt PShellLab4\DOCS
Copy-Item PShellLab4\TEXT\8.txt PShellLab4\DOCS

# Перемещение файлов
Move-Item PShellLab4\TEXT\3.txt PShellLab4\VIDEO
Move-Item PShellLab4\TEXT\4.txt PShellLab4\VIDEO
Move-Item PShellLab4\TEXT\5.txt PShellLab4\VIDEO

# Переименование файлов
Rename-Item PShellLab4\DOCS\7.txt "07may_7.txt"

# Удаление каталогов
Remove-Item PShellLab4\DOCX -Recurse
Remove-Item PShellLab4\TXT -Recurse

# Отображение даты и времени
Get-Date

# Справка PowerShell
Get-Help *

# Просмотр GRAPH и TEXT
Get-ChildItem PShellLab4\GRAPH
Get-ChildItem PShellLab4\TEXT

# Переход в GRAPH
Set-Location PShellLab4\GRAPH

# Очистка GRAPH
Remove-Item * -Recurse

# Итоговый просмотр
Get-ChildItem PShellLab4 -Recurse

# Удаление PShellLab4
Remove-Item PShellLab4 -Recurse



# =========================
# ЗАДАНИЕ 2. ПРОЦЕССЫ
# =========================

# Список процессов
Get-Process

# Информация о explorer
Get-Process explorer | Select-Object Id, CPU, WorkingSet

# Топ-5 процессов по памяти
Get-Process |
Sort-Object WorkingSet -Descending |
Select-Object -First 5

# Экспорт в CSV
Get-Process |
Sort-Object WorkingSet -Descending |
Select-Object -First 5 |
Export-Csv top_processes.csv -NoTypeInformation

# Запуск блокнота
Start-Process notepad

# Завершение блокнота
Stop-Process -Name notepad

# Справка по Get-Process
Get-Help Get-Process



# =========================
# ЗАДАНИЕ 3. СЛУЖБЫ
# =========================

# Список служб
Get-Service

# Только запущенные службы
Get-Service | Where-Object {$_.Status -eq "Running"}

# Проверка службы Spooler
Get-Service Spooler

# Остановка службы
Stop-Service Spooler

# Зависимости службы
Get-Service Spooler |
Select-Object -ExpandProperty ServicesDependedOn

# Справка по Get-Service
Get-Help Get-Service
